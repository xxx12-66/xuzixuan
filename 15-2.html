一、条件结构

① if / else if / else 条件判断 
// 根据分数输出等级
var score = 85;
if (score >= 90) {
    console.log("优秀");
} else if (score >= 75) {
    console.log("良好");
} else if (score >= 60) {
    console.log("及格");
} else {
    console.log("不及格");
}
② switch / case 多分支 
var day = 3;
switch (day) {
    case 1: console.log("周一"); break;
    case 2: console.log("周二"); break;
    case 3: console.log("周三"); break;
    default: console.log("其他");
}
③ 三元运算符 condition ? a : b 简写 if/else 
var age = 20;
var tip = (age >= 18) ? "成年人" : "未成年人";
console.log(tip);

var num = -5;
var absVal = (num >= 0) ? num : -num;
console.log("绝对值：" + absVal);

二、循环结构

① for 循环（最常用） 计数循环 
// 输出 1 到 n
for (var i = 1; i <= 5; i++) {
    console.log(i);
}

// 求 1 到 n 的和
var sum = 0;
for (var j = 1; j <= 5; j++) {
    sum += j;
}
console.log("sum = " + sum);
② while 循环 条件为真才执行 
// 模拟：每次减 3，直到 ≤ 0
var n = 10;
while (n > 0) {
    console.log("n = " + n);
    n -= 3;
}
console.log("结束，n = " + n);
③ do...while 循环 至少执行一次 
// 先执行一次，再判断条件
var x = 1;
do {
    console.log("x = " + x);
    x *= 2;
} while (x < 20);
④ for...of 遍历数组 ES6 新语法 
var fruits = ["苹果", "香蕉", "橙子", "葡萄"];
for (var fruit of fruits) {
    console.log(fruit);
}
⑤ for...in 遍历对象属性 枚举键名 
var person = { name: "徐梓轩", age: 20, school: "XX大学" };
for (var key in person) {
    console.log(key + " = " + person[key]);
}
⑥ break 和 continue 循环控制 
// break：遇到 7 就停止整个循环
for (var i = 1; i <= 10; i++) {
    if (i === 7) { console.log("break!"); break; }
    console.log(i);
}

// continue：跳过偶数，只输出奇数
for (var j = 1; j <= 10; j++) {
    if (j % 2 === 0) continue;
    console.log(j);
}
