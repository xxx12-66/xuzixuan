核心代码

// 列表接口：limit 控制条数，skip 控制分页起点
var url = 'https://dummyjson.com/products?limit=8';

async function loadList() {
    var res  = await fetch(url);
    var data = await res.json();

    // 兼容两种返回：{products:[...]} 或直接 [...]
    var list = Array.isArray(data) ? data : data.products;

    // 遍历拼接 HTML：每个卡片用 a 标签跳详情，并把 id 放到地址栏
    list.forEach(function (p) {
        html += '<a href="17-3-detail.html?id=' + p.id + '">...</a>';
    });
}

要点：① 分页靠 ?limit=8&skip=8 两个参数实现；② 卡片用 <a> 承载， 跳转时把 id 拼进 URL；③ 详情页用 location.search 取回这个 id 再请求单条数据。
