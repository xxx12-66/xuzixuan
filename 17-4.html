核心代码：一个通用的 fetch 请求函数

async function request(method, url, headers, body) {
    var options = { method: method, headers: headers };
    if (method !== 'GET' && method !== 'DELETE' && body) {
        options.body = JSON.stringify(body);   // 对象 → JSON 字符串
    }

    var res = await fetch(url, options);
    var text = await res.text();          // 先拿纯文本，避免非 JSON 直接报错

    try {
        var data = JSON.parse(text);       // 再尝试解析成对象
        show(res.status, data);
    } catch (e) {
        show(res.status, text);          // 解析失败就按文本展示
    }
}

要点：① fetch(url, {method, headers, body}) 的第二个参数就是请求配置，GET/DELETE 不能带 body； ② body 必须是字符串，对象要先 JSON.stringify()； ③ HTTP 404/500 不会触发 catch，必须用 res.ok 或 res.status 判断； ④ 只有网络失败 / 跨域被拒 / 请求超时才会进入 catch（本页用 AbortController 实现 15 秒超时）。
