17-5 · JavaScript 两种定时器

setTimeout（延时一次）与 setInterval（循环执行）—— 点击按钮观察执行过程

一、setTimeout —— 延时执行一次

语法：var id = setTimeout(回调函数, 延迟毫秒, 参数...)； 返回一个数字 id，可用 clearTimeout(id) 在触发前取消。

① 基本用法：延时后执行一次 setTimeout 
var timerId = setTimeout(function () {
    console.log("3 秒后我才执行，而且只执行一次");
}, 3000);

// 在触发前取消
clearTimeout(timerId);
ms
② 递归 setTimeout 实现倒计时 循环的经典写法 
var n = 5;
function tick() {
    console.log(n + " ...");
    n--;
    if (n > 0) {
        setTimeout(tick, 1000);   // 自己调用自己 → 形成循环
    } else {
        console.log("出发！");
    }
}
tick();
00
③ 延时设为 0：也不会立刻执行 异步 / 事件循环 
console.log("1. 同步代码");
setTimeout(function () { console.log("3. 定时器回调（宏任务）"); }, 0);
console.log("2. 同步代码结束");
// 结果顺序：1 → 2 → 3，定时器回调永远排在同步代码之后

二、setInterval —— 每隔一段时间重复执行

语法：var id = setInterval(回调函数, 间隔毫秒)； 必须手动 clearInterval(id) 才会停止，否则会一直运行下去。

① 每秒计数器 setInterval 
var count = 0;
var id = setInterval(function () {
    count++;
    console.log("已经过了 " + count + " 秒");
    if (count >= 10) clearInterval(id);   // 到条件就停止
}, 1000);
ms
② 实时电子时钟 每秒刷新 
var clockId = setInterval(function () {
    var now = new Date();
    var h = pad(now.getHours());
    var m = pad(now.getMinutes());
    var s = pad(now.getSeconds());
    clockEl.textContent = h + ":" + m + ":" + s;
}, 1000);
--:--:--
③ 进度条动画（可暂停 / 继续 / 重置） 定时器 + DOM 
var percent = 0;
var barId = setInterval(function () {
    percent += 2;
    bar.style.width = percent + "%";
    if (percent >= 100) clearInterval(barId);   // 满了就停
}, 100);
0%

三、setTimeout 与 setInterval 对比

对比项setTimeoutsetInterval
执行次数延迟后执行 一次每隔固定时间 重复执行，直到被清除
语法setTimeout(fn, ms)setInterval(fn, ms)
取消方法clearTimeout(id)clearInterval(id)
返回值都是一个数字 定时器 id，浏览器环境下从 1 开始递增（两者 id 各自独立编号）
实现循环需在回调内递归再次调用自身天然循环
间隔准确性递归方式保证两次执行之间至少间隔 ms若回调耗时 > 间隔，会造成任务堆积
常见用途防抖 debounce、延时提示、延时跳转、动画收尾轮询、时钟、轮播图、进度刷新
注意：① 定时器回调里的 this 默认指向 window（严格模式下为 undefined），需要访问外部 this 请用箭头函数或先存 var self = this； ② 页面卸载前记得 clearInterval，否则定时器一直占用内存； ③ 时间参数单位是毫秒（1000ms = 1 秒），且只表示"最快什么时候执行"，不保证精确。

四、经典陷阱：循环里的 var + 定时器

① 为什么输出的都是 5？ 闭包 
// ❌ var 是函数作用域，5 个回调共享同一个 i，执行时循环早已结束
for (var i = 0; i < 5; i++) {
    setTimeout(function () { console.log(i); }, 100);
}   // 输出：5 5 5 5 5

// ✅ 改法一：let 是块级作用域，每次循环都是新的 i
for (let i = 0; i < 5; i++) {
    setTimeout(function () { console.log(i); }, 100);
}   // 输出：0 1 2 3 4

// ✅ 改法二：用 setTimeout 的第三个参数把值传进去
for (var j = 0; j < 5; j++) {
    setTimeout(function (n) { console.log(n); }, 100, j);
}

五、综合实战：秒表（setInterval + 清除）

开始 / 暂停 / 继续 / 重置 综合练习 
var ms = 0, running = false, swId = null;

function start() {
    if (running) return;
    running = true;
    swId = setInterval(function () {
        ms += 100;                       // 每 100ms 累加一次
        showEl.textContent = format(ms);
    }, 100);
}
function pause() { running = false; clearInterval(swId); }   // 清除即暂停
function reset() { pause(); ms = 0; showEl.textContent = "00:00.0"; }
00:00.0
