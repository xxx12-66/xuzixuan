一、什么是 JSON

JSON（JavaScript Object Notation）是一种轻量级的数据交换格式，本质是"符合特定语法的字符串"。 它独立于语言，常被用于前后端数据传输与配置文件。

① 一份典型的 JSON 数据 对象 + 数组 + 嵌套 

    


    

      
        
语法规则说明示例

      
      
        

          键名必须双引号
          JSON 的键必须用 "" 包裹，不能省略、不能用单引号
          ✓ {"name":"张三"}   ✗ {name:'张三'}
        

        

          字符串必须双引号
          字符串值只能用双引号，不能用单引号或反引号
          ✓ "北京"   ✗ '北京'
        

        

          支持的数据类型
          对象、数组、字符串、数字、布尔、null
          {"age":20,"vip":true,"tel":null}
        

        

          不支持的类型
          不能出现 undefined、函数、日期对象、注释
          ✗ {"fn":function(){}}
        

        

          末尾不能多余逗号
          最后一个成员后面不能再写逗号
          ✗ {"a":1,}
        

        

          两种结构
          对象 { }（键值对集合）与数组 [ ]（有序列表），可任意嵌套
          [{"id":1},{"id":2}]
        

      
    

  


  
  

    
二、JSON.stringify() —— 对象转 JSON 字符串

    

      语法：JSON.stringify(value, replacer?, space?)。第三个参数控制缩进，用于"格式化输出（美化）"。
    


    

      

        ① 对象 → JSON 字符串（含缩进参数对比）
        序列化
      


var student = { name: "徐梓轩", age: 20, score: 92.5, vip: true };

// 1. 不带缩进：压缩成一行（网络传输用）
JSON.stringify(student);
// 2. 缩进 2 个空格：格式化（调试查看用）
JSON.stringify(student, null, 2);

// typeof 检测：结果是字符串，不再是对象
console.log(typeof JSON.stringify(student));  // "string"

      

        
        
        
        
      

      

    


    

      

        ② 数组 / 嵌套对象 同样适用
        数组序列化
      


var list = [
  { id: 1, name: "HTML",  hour: 16 },
  { id: 2, name: "CSS",   hour: 24 },
  { id: 3, name: "JavaScript", hour: 32 }
];

console.log(JSON.stringify(list, null, 2));

      

        
        
      

      

    

  


  
  

    
三、JSON.parse() —— JSON 字符串转对象

    

      语法：JSON.parse(text)。把后端返回的 JSON 字符串"还原"成 JS 对象/数组，之后才能用
      obj.name、arr[0] 等方式取值。
    


    

      

        ① 字符串 → 对象，并读取属性
        反序列化
      


var text = '{"name":"徐梓轩","age":20,"skills":["HTML","CSS","JS"]}';

var obj = JSON.parse(text);        // 字符串 → 对象
console.log(obj.name);            // "徐梓轩"   点语法取值
console.log(obj["age"]);          // 20         中括号取值
console.log(obj.skills[1]);       // "CSS"      数组下标取值

      

        

          
        

        
        
      

      

    


    

      

        ② 非法 JSON 会报错 → 用 try...catch 捕获
        异常处理
      


var bad = "{name:'徐梓轩',}";   // 键没加双引号 + 末尾多余逗号

try {
    var o = JSON.parse(bad);
    console.log("解析成功", o);
} catch (e) {
    console.log("解析失败：" + e.message);   // 不写 try 会中断整个脚本
}

      

        

          
        

        
        
        
      

      

    

  


  
  

    
四、遍历 JSON 数据

    
解析后的 JSON 就是普通的 JS 对象/数组，数组用 for / forEach，对象用 for...in。


    

      

        ① 遍历 JSON 数组 + 统计
        for / forEach
      


var json = '[{"name":"张三","score":88},{"name":"李四","score":95},{"name":"王五","score":72}]';
var arr = JSON.parse(json);

var sum = 0, max = 0;
for (var i = 0; i < arr.length; i++) {
    console.log(arr[i].name + "：" + arr[i].score);
    sum += arr[i].score;
    if (arr[i].score > max) max = arr[i].score;
}
console.log("平均分：" + (sum / arr.length).toFixed(1) + "，最高分：" + max);

      

        
        
      

      

    


    

      

        ② for...in 遍历 JSON 对象的键
        枚举键名
      


var obj = JSON.parse('{"title":"高级网页设计","credit":3,"teacher":"徐老师"}');
for (var key in obj) {
    console.log(key + " = " + obj[key]);
}
