执行日志（Promise 状态）

核心代码

// 用 Promise 包装 setTimeout：延时结束后 resolve
function delay(ms, name) {
    return new Promise(function (resolve) {
        console.log(name + ' 开始（Promise pending）');
        setTimeout(function () {
            resolve(name + ' 完成');      // → fulfilled
        }, ms);
    });
}

// async 函数 + await：像写同步代码一样写异步流程
async function makeDumplings() {
    await delay(6000, '买原料');          // 串行：上一个完成才走下一个
    await delay(2000, '取原料');

    // 并行：同时和面、和馅，两个都完成再往下走
    await Promise.all([
        delay(4000, '和面'),
        delay(3000, '和馅')
    ]);

    await delay(5000, '包饺子');
    await delay(2000, '吃饺子');
}

makeDumplings().then(function () { console.log('全部完成 ✅'); });

要点：① await 会等待 Promise 变成 fulfilled 才执行下一行，因此顺序流程写起来像同步代码； ② 需要"同时做"的任务必须先各自创建 Promise，再用 Promise.all 一起等待—— 写成两个 await 会变成串行（耗时 4s + 3s = 7s，而并行只需 4s）； ③ 如果某个环节可能失败，用 try...catch 或 Promise.allSettled 处理； ④ 本页的"播放速度"只是把毫秒数除以倍速，不改变代码结构。
