核心代码

// 用 Promise 包装 setTimeout：延时结束后 resolve
function delay(ms, name) {
    return new Promise(function (resolve) {
        setTimeout(function () { resolve(name); }, ms);
    });
}

async function makeDumplings() {
    // ① 竞速：线上 6s / 线下 8s 同时开始，谁先成功就用谁
    var winner = await Promise.any([
        delay(6000, '线上购买'),
        delay(8000, '线下购买')
    ]);                                  // 较慢的那个不阻塞流程

    await delay(2000, '取原料');       // ② 串行

    // ③ 并行：和面 + 和馅，两个都完成才继续
    await Promise.all([
        delay(4000, '和面'),
        delay(3000, '和馅')
    ]);

    await delay(5000, '包饺子');       // ④ 串行
    await delay(3000, '煮饺子');       // ⑤ 串行
    await delay(2000, '吃饺子');       // ⑥ 串行
}

三个组合方法的分工： Promise.all —— 全部成功才继续（和面 + 和馅，用于"都要完成"）； Promise.any —— 第一个成功就继续（线上 / 线下买原料，用于"完成任何一个即可"）； Promise.race —— 第一个"敲定"就继续（不论成功失败）。
要点：① 竞速的两个任务必须先同时创建，再交给 Promise.any 等待，写成两个 await 就成了串行； ② 落败的任务并不会被取消，只是结果被忽略（本页日志会显示它最终也完成了）； ③ 若所有任务都失败，Promise.any 才抛 AggregateError，可用 try...catch 兜底； ④ "播放速度"只是把毫秒数除以倍速，不改变代码结构。
